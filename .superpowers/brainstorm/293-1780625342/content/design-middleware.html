middleware/monitor_mw.py
class PerformanceMonitor(BaseCallbackHandler):
    """全链路性能监控"""

    def on_chain_start(self, ...):
        self.start_time = time.time()
        self.tool_calls = 0
        self.total_tokens = 0

    def on_tool_start(self, ...):
        self.tool_calls += 1

    def on_llm_end(self, response, **kwargs):
        usage = response.llm_output.get("token_usage", {})
        self.total_tokens += usage.get("total_tokens", 0)

    def on_chain_end(self, outputs, **kwargs):
        elapsed = time.time() - self.start_time
        self._report({
            "latency_ms": elapsed * 1000,
            "tool_calls": self.tool_calls,
            "total_tokens": self.total_tokens,
        })
middleware/prompt_switch.py — 动态 Prompt 切换
from src.agent.prompt import PRESALE_PROMPT, AFTERSALE_PROMPT, MAINTENANCE_PROMPT, SYSTEM_PROMPT_BASE

# 关键词 → 场景映射
SCENARIO_KEYWORDS = {
    "presale": ["购买", "多少钱", "哪个型号", "对比", "推荐", "有什么区别", "参数"],
    "aftersale": ["坏了", "报错", "故障", "不工作", "报修", "维修", "E0", "E1", "错误代码"],
    "maintenance": ["保养", "清理", "更换滤网", "耗材", "怎么清洗", "多久换"],
}

def detect_scenario(user_query: str) -> str:
    """根据用户问题自动检测场景"""
    for scenario, keywords in SCENARIO_KEYWORDS.items():
        if any(kw in user_query for kw in keywords):
            return scenario
    return "default"

def get_prompt_for_scenario(scenario: str) -> str:
    prompts = {
        "presale": PRESALE_PROMPT,
        "aftersale": AFTERSALE_PROMPT,
        "maintenance": MAINTENANCE_PROMPT,
        "default": SYSTEM_PROMPT_BASE,
    }
    return prompts.get(scenario, SYSTEM_PROMPT_BASE)

📦 中间件注册

src/middleware/__init__.py — 统一中间件链
from src.middleware.logging_mw import AuditLoggingHandler
from src.middleware.monitor_mw import PerformanceMonitor

# 中间件链 — 按顺序执行
MIDDLEWARE_CHAIN = [
    AuditLoggingHandler(),     # 1. 先审计
    PerformanceMonitor(),      # 2. 再监控
]

# Agent 配置时注入 callbacks
def build_agent_config(callbacks=None):
    if callbacks is None:
        callbacks = MIDDLEWARE_CHAIN
    return {"callbacks": callbacks}

📋 设计要点

实现机制 LangChain BaseCallbackHandler — 原生钩子，无需侵入 Agent 代码
审计日志 每次 LLM 调用 + 工具调用写一条 JSONL，存入 logs/audit.jsonl
监控指标 首字延迟、总耗时、token 用量、工具调用次数 — 结构化为日志输出
Prompt 切换 关键词匹配检测场景（简单高效），后续可升级为 LLM 意图分类
