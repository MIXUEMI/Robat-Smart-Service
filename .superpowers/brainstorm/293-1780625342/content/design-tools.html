🔧 关键工具设计细节

search_knowledge_base (最核心)
@tool
def search_knowledge_base(query: str) -> str:
    """搜索产品知识库，获取相关文档。
    用于回答产品功能、使用方法、故障排查等问题。"""

    retriever = get_retriever()
    docs = retriever.retrieve(query, top_k=4)

    if not docs:
        return "未找到相关知识。"

    result = []
    for i, doc in enumerate(docs):
        result.append(
            f"[来源{i+1}: {doc.metadata['source']}]\n"
            f"{doc.page_content}"
        )

    return "\n\n---\n\n".join(result)
