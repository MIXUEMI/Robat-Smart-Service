💻 核心代码设计

app.py — Streamlit 主入口
import streamlit as st
import requests
import json
import uuid

st.set_page_config(
    page_title="小扫 · 智能客服",
    page_icon="🤖",
    layout="wide",
)

# ── 侧边栏 ──
with st.sidebar:
    st.title("🤖 小扫 · 智能客服")
    st.caption("扫地机器人专属助手")

    if st.button("✨ 新对话", use_container_width=True):
        st.session_state.session_id = str(uuid.uuid4())
        st.session_state.messages = []
        st.rerun()

    # 显示系统状态
    with st.expander("📊 系统状态"):
        try:
            health = requests.get("http://localhost:8000/api/v1/health").json()
            st.success(f"模型: {health['model']}")
            st.info(f"向量库: {health['vectordb_status']}")
        except:
            st.error("服务未连接")

# ── 初始化会话 ──
if "session_id" not in st.session_state:
    st.session_state.session_id = str(uuid.uuid4())
if "messages" not in st.session_state:
    st.session_state.messages = []

# ── 渲染历史消息 ──
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])

# ── 输入框 ──
if prompt := st.chat_input("输入你的问题..."):
    # 显示用户消息
    st.session_state.messages.append({"role": "user", "content": prompt})
    with st.chat_message("user"):
        st.markdown(prompt)

    # 调用流式 API
    with st.chat_message("assistant"):
        placeholder = st.empty()
        full_response = ""

        with requests.post(
            "http://localhost:8000/api/v1/chat/stream",
            json={"session_id": st.session_state.session_id, "query": prompt},
            stream=True,
        ) as resp:
            for line in resp.iter_lines():
                if not line:
                    continue
                line = line.decode()
                if line.startswith("data:"):
                    data = json.loads(line[5:].strip())
                elif line.startswith("event:"):
                    event = line[6:].strip()
                    continue

                if event == "token":
                    full_response += data
                    placeholder.markdown(full_response + "▌")
                elif event == "tool_start":
                    with st.status(f"🔧 调用工具: {data}", expanded=False):
                        pass
                elif event == "done":
                    placeholder.markdown(full_response)

        st.session_state.messages.append({
            "role": "assistant", "content": full_response
        })

📱 交互设计要点

流式渲染 st.empty() placeholder + SSE 逐 token 更新，模拟打字效果
工具可见 st.status() 展示工具调用过程（可折叠），透明化 Agent 行为
会话管理 st.session_state — session_id + messages 列表，保持对话上下文
健康检查 侧边栏轮询 FastAPI /health，实时展示系统状态
错误处理 API 超时/断连时优雅降级，显示错误提示而非崩溃
