⚙️ Agent 核心机制

src/agent/agent.py — 基于 LangChain create_react_agent
from langgraph.prebuilt import create_react_agent
from langchain_core.prompts import ChatPromptTemplate

class CustomerServiceAgent:
    """扫地机器人智能客服 Agent"""

    def __init__(self, tools: list, llm, system_prompt: str):
        self.tools = tools
        self.llm = llm.bind_tools(tools)  # 绑定工具到模型
        self.system_prompt = system_prompt

        # 使用 LangGraph 的 create_react_agent
        self.agent = create_react_agent(
            model=self.llm,
            tools=self.tools,
            prompt=self.system_prompt,
        )

    async def astream(self, messages: list):
        """异步流式执行 — 逐 token 返回"""
        async for event in self.agent.astream_events(
            {"messages": messages},
            version="v2",
        ):
            yield self._parse_event(event)

    def invoke(self, messages: list) -> dict:
        """同步执行 — 返回完整结果"""
        return self.agent.invoke({"messages": messages})

💬 对话记忆管理

src/agent/memory.py
from langchain_core.chat_history import (
    InMemoryChatMessageHistory,
)
from langchain_core.runnables.history import (
    RunnableWithMessageHistory,
)

class ConversationManager:
    """会话管理 — 内存存储，支持多会话"""

    def __init__(self):
        self._store: dict[str, InMemoryChatMessageHistory] = {}

    def get_history(self, session_id: str):
        if session_id not in self._store:
            self._store[session_id] = (
                InMemoryChatMessageHistory()
            )
        return self._store[session_id]

    def clear(self, session_id: str):
        if session_id in self._store:
            del self._store[session_id]

    def get_recent_messages(
        self, session_id: str, n: int = 10
    ) -> list:
        history = self.get_history(session_id)
        return history.messages[-n:]

🔑 关键设计决策

Agent 框架 langgraph.prebuilt.create_react_agent — 比旧版 langchain.agents 更稳定，原生支持 streaming
工具绑定 llm.bind_tools(tools) — DeepSeek 支持原生 function calling
流式输出 astream_events — 逐 token 流式返回，支持 on_tool_start/on_tool_end 事件
消息历史 InMemoryChatMessageHistory — 会话级内存存储，后续可切换到 Redis/DB
最大迭代 Agent 最多执行 10 轮 Thought-Action，防止无限循环
